@app.route('/import-fb-accounts', methods=['POST'])
def import_fb_accounts():

    file = request.files['file']

    df = pd.read_excel(file)

    for _, row in df.iterrows():

        account = FBAccount(

            account_name=row.get('account_name'),
            email=row.get('email'),
            phone=row.get('phone'),

            fb_id=row.get('fb_id'),
            password=row.get('password'),

            purchase_date=row.get('purchase_date'),
            purchase_cost=row.get('purchase_cost', 0),

            status=row.get('status', 'Active'),
            notes=row.get('notes')
        )

        db.session.add(account)

    db.session.commit()

    flash('Accounts Imported Successfully!', 'success')

    return redirect('/fb-accounts')


@app.route('/export-fb-accounts')
def export_fb_accounts():

    accounts = FBAccount.query.all()

    data = []

    for acc in accounts:

        data.append({

            'account_name': acc.account_name,
            'email': acc.email,
            'phone': acc.phone,

            'fb_id': acc.fb_id,
            'password': acc.password,

            'purchase_date': acc.purchase_date,
            'purchase_cost': acc.purchase_cost,

            'status': acc.status,
            'notes': acc.notes
        })

    df = pd.DataFrame(data)

    file_name = 'fb_accounts.xlsx'

    df.to_excel(file_name, index=False)

    return send_file(file_name, as_attachment=True)
